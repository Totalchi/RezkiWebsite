Supabase not configured Dashboard is in demo mode. To connect real leads:
  1. Create a Supabase project at supabase.com
  2. Run the SQL below to create the leads table
  3. Add your supabaseUrl and supabaseAnonKey to index.html in RM_AUTH_CONFIG
  4. Do the same here in admin.html in ADMIN_CONFIG
CREATE TABLE leads ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, created_at TIMESTAMPTZ DEFAULT NOW(), type TEXT, -- 'book' | 'quote' | 'invoice' lang TEXT, name TEXT, email TEXT, phone TEXT, company TEXT, address TEXT, services TEXT, property_type TEXT, timing TEXT, notes TEXT, booking_slot TEXT, status TEXT DEFAULT 'new' ); -- Row-level security: only authenticated users can read/update ALTER TABLE leads ENABLE ROW LEVEL SECURITY; CREATE POLICY "Admins only" ON leads USING (auth.role() = 'authenticated') WITH CHECK (auth.role() = 'authenticated');
